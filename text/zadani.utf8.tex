\nadepsanaSranka{Zadání}
 \begin{enumerate}
       \item Prostudujte problematiku S-boxů v kryptografii. Dále prostudujte evoluční výpočetní techniky, jako jsou genetické algoritmy, algoritmy EDA a kartézské genetické programování.
       \item Navrhněte program pro evoluční návrh S-boxů pomocí jednotlivých metod.
       \item Navržený program implementujte ve vhodném jazyce a prostředí.
       \item Porovnejte jednotlivé přístupy na zvolených příkladech a zhodnoťte dosažené výsledky.
 \end{enumerate}
